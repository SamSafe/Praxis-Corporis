extends Control
func _ready() -> void:
	call_deferred("run")
func run() -> void:
	var button := MenuButton.new()
	add_child(button)
	var popup := button.get_popup()
	popup.add_item("Example",0)
	for i in range(12):
		if "--settle" in OS.get_cmdline_user_args():
			await get_tree().process_frame
		button.show_popup()
		popup.hide()
	print("PASS minimal popup cycles")
	get_tree().quit()
